# Data Models: Vision-Language-Action (VLA) Robotics

## Core Entities

### 1. VoiceCommand
Represents a spoken instruction from a human user, including audio data, transcribed text, and semantic intent.

```typescript
interface VoiceCommand {
  id: string;
  timestamp: Date;
  audioData?: ArrayBuffer;  // Raw audio data (optional, may be deleted after processing)
  transcription: string;    // Transcribed text from audio
  confidence: number;       // Confidence score of transcription (0-1)
  intent: string;           // Parsed semantic intent (e.g. "NAVIGATE_TO_LOCATION", "GRASP_OBJECT")
  rawAudioPath?: string;    // File path to audio recording (for debugging)
  metadata: {
    sourceDevice: string;   // Device ID where command originated
    audioFormat: string;    // Format of audio (e.g. "wav", "mp3")
    language: string;       // Detected language
    noiseLevel: number;     // Ambient noise estimate
  };
}
```

### 2. ActionPlan
Sequence of ROS 2 actions generated by the LLM to fulfill a command, including navigation, perception, and manipulation steps.

```typescript
interface ActionPlan {
  id: string;
  commandId: string;        // Reference to originating VoiceCommand
  createdAt: Date;
  actionSequence: ActionStep[];
  status: "PLANNED" | "IN_PROGRESS" | "COMPLETED" | "FAILED" | "CANCELLED";
  estimatedDuration: number; // Estimated time to complete in seconds
  priority: number;         // Priority level (0-10)
  constraints: PlanConstraint[]; // Kinematic, safety, or environmental constraints
  metadata: {
    llmModelUsed: string;   // Model that generated the plan (e.g. "gpt-4-turbo")
    promptTokens: number;   // Number of tokens in prompt
    completionTokens: number; // Number of tokens in response
  };
}

interface ActionStep {
  id: string;
  type: "NAVIGATION" | "PERCEPTION" | "MANIPULATION" | "WAIT" | "CONDITIONAL";
  description: string;      // Human-readable description of the action
  parameters: Record<string, any>; // Action-specific parameters (e.g. target pose, object ID)
  timeout: number;          // Max time to complete this step in seconds
  precondition?: string;    // Condition that must be met before this step
  fallbackAction?: ActionStep; // What to do if this step fails
  estimatedDuration: number; // Estimated duration in seconds
}

interface PlanConstraint {
  type: "KINEMATIC" | "SAFETY" | "ENVIRONMENTAL" | "RESOURCE";
  description: string;
  severity: "WARNING" | "ERROR";  // Whether constraint violation prevents execution
}
```

### 3. RobotState
Current configuration of the robot including position, orientation, joint angles, and task progress.

```typescript
interface RobotState {
  id: string;
  timestamp: Date;
  pose: Pose;               // Position and orientation in world frame
  jointStates: JointState[]; // Current joint angles and velocities
  batteryLevel: number;     // Battery charge percentage (0-100)
  operationalStatus: "ACTIVE" | "STANDBY" | "ERROR" | "CHARGING" | "MAINTENANCE";
  activeTasks: string[];    // IDs of currently executing tasks
  sensors: SensorData[];    // Current readings from all sensors
  gripperState: GripperState; // State of end effectors
  connected: boolean;       // Whether robot is connected to system
  capabilities: RobotCapability[]; // List of available capabilities
}

interface Pose {
  position: Point3D;
  orientation: Quaternion;  // In quaternions (x, y, z, w)
}

interface Point3D {
  x: number;
  y: number;
  z: number;
}

interface Quaternion {
  x: number;
  y: number;
  z: number;
  w: number;
}

interface JointState {
  name: string;
  position: number;         // Joint angle in radians
  velocity: number;         // Joint velocity in rad/s
  effort: number;           // Applied torque/force
}

interface SensorData {
  sensorId: string;
  type: "CAMERA_RGB" | "CAMERA_DEPTH" | "LIDAR" | "IMU" | "FORCE_TORQUE";
  data: any;                // Sensor-specific data format
  timestamp: Date;
}

interface GripperState {
  left: {
    position: number;       // Position in meters (0=closed, 1=open)
    force: number;          // Applied force in Newtons
    objectHeld: boolean;
    objectId?: string;      // ID of held object if any
  };
  right: {
    position: number;
    force: number;
    objectHeld: boolean;
    objectId?: string;
  };
}

interface RobotCapability {
  type: "NAVIGATION" | "MANIPULATION" | "PERCEPTION" | "COMMUNICATION";
  name: string;
  description: string;
  enabled: boolean;
}
```

### 4. EnvironmentalMap
Representation of the environment including obstacles, objects, and navigable areas.

```typescript
interface EnvironmentalMap {
  id: string;
  version: string;          // Map version identifier
  createdAt: Date;
  origin: Pose;             // Origin of map coordinate frame
  resolution: number;       // Resolution of occupancy grid in meters per cell
  occupancyGrid: OccupancyGrid; // Static map data
  dynamicObjects: DetectedObject[]; // Moving objects detected since map creation
  navigationCosts: NavigationCostMap; // Costs for path planning
  metadata: {
    source: string;         // System that created map (e.g. "SLAM", "manual")
    updateFrequency: number; // How often map is updated (Hz)
    coordinateFrame: string; // TF frame name (e.g. "map")
  };
}

interface OccupancyGrid {
  width: number;            // Width in cells
  height: number;           // Height in cells
  data: number[];           // Array of values (-1: unknown, 0: free, 100: occupied)
}

interface NavigationCostMap {
  width: number;
  height: number;
  data: number[];           // Cost values (higher means more costly to traverse)
}
```

### 5. DetectedObject
Identified items in the environment with properties like position, class, and affordances for manipulation.

```typescript
interface DetectedObject {
  id: string;
  classLabel: string;       // Semantic class (e.g. "chair", "cup", "person")
  confidence: number;       // Detection confidence (0-1)
  boundingBox: BoundingBox3D; // 3D bounding box
  center: Point3D;          // Center position in world coordinates
  dimensions: Point3D;      // Size in x, y, z dimensions
  pose: Pose;               // Orientation as well as position
  affordances: ObjectAffordance[]; // Possible interactions
  color: RGBColor;          // Dominant color
  material: string;         // Material type (e.g. "metal", "plastic", "fabric")
  isGraspable: boolean;     // Whether object can be grasped
  graspPoints: GraspPoint[]; // Valid points for grasping
  isMoveable: boolean;      // Whether object can be moved by robot
  weightEstimate: number;   // Estimated weight in kg
  timestamp: Date;          // When object was detected
  sourceSensor: string;     // Sensor that detected the object
  trackingId?: string;      // ID for tracking across frames
}

interface BoundingBox3D {
  center: Point3D;
  dimensions: Point3D;
  corners: Point3D[];       // 8 corner points
}

interface ObjectAffordance {
  type: "grasp" | "push" | "lift" | "move" | "pour"; // Type of interaction
  position: Point3D;        // Position where action applies
  direction?: Vector3D;     // Direction of action (if applicable)
  forceRange?: { min: number; max: number }; // Force limits
  metadata: {
    confidence: number;
    method: string;         // How affordance was determined
  };
}

interface RGBColor {
  r: number;                // Red value (0-255)
  g: number;                // Green value (0-255)
  b: number;                // Blue value (0-255)
}

interface GraspPoint {
  position: Point3D;        // Position of grasp point
  orientation: Quaternion;  // Preferred orientation of gripper
  stabilityScore: number;   // How stable this grasp is expected to be (0-1)
  graspType: "power" | "precision" | "pinch"; // Type of grasp
}
```

### 6. ExecutionContext
Contextual information during task execution including intermediate results, failures, and adaptations.

```typescript
interface ExecutionContext {
  id: string;
  planId: string;           // Reference to the executing ActionPlan
  startTime: Date;
  currentStepIndex: number; // Index of currently executing step
  completedSteps: number[]; // Indices of completed steps
  failedSteps: FailedStep[]; // Steps that failed during execution
  intermediateResults: Record<string, any>; // Results from executed steps
  adaptationHistory: AdaptationEvent[]; // Changes made to plan during execution
  currentStatus: "INITIALIZING" | "RUNNING" | "PAUSED" | "FAILED" | "COMPLETED";
  progress: number;         // Progress percentage (0-100)
  estimatedTimeRemaining: number; // Estimated seconds until completion
  safetyViolations: SafetyViolation[]; // Safety issues encountered
  replanRequests: ReplanRequest[]; // Requests for plan modifications
}

interface FailedStep {
  stepId: string;
  reason: string;           // Description of why step failed
  recoveryAttempted: string; // Action taken to recover
  timestamp: Date;
}

interface AdaptationEvent {
  type: "STEP_SKIP" | "STEP_REPEAT" | "STEP_MODIFY" | "SUBGOAL_ADD";
  originalStepId: string;
  modifiedStep?: ActionStep; // New version of step (for STEP_MODIFY)
  reason: string;           // Why adaptation was needed
  timestamp: Date;
}

interface SafetyViolation {
  severity: "WARNING" | "ERROR" | "CRITICAL";
  description: string;
  actionTaken: string;      // What the system did in response
  timestamp: Date;
}

interface ReplanRequest {
  reason: "OBSTACLE_DETECTED" | "TARGET_MOVED" | "GRASP_FAILED" | "PATH_BLOCKED";
  suggestedModification: string; // Description of needed change
  priority: "LOW" | "MEDIUM" | "HIGH";
  timestamp: Date;
}
```

## Relationships Between Entities

```
VoiceCommand -(generates)> ActionPlan -(executes in context)> ExecutionContext
ActionPlan -(references)> EnvironmentalMap -(contains)> DetectedObject
RobotState -(contains)> Pose -(relates to)> EnvironmentalMap
ExecutionContext -(updates)> RobotState
DetectedObject -(has)> ObjectAffordance
ActionPlan -(contains)> ActionStep +(requires)> DetectedObject
ExecutionContext -(records)> FailedStep +(references)> ActionStep
```
